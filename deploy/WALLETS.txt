AFTER successful release, explicitly enable the wallets in the EXTERNAL config.
This opt-in is default-off and affects manual transfers only, not card or SMTP.
The following creates a private backup and atomically updates the JSON. It does
not change SECRET_KEY or any database records, and prints no secret values.

python3.13 - <<'PY'
import json, os, shutil, tempfile
from datetime import datetime, timezone
from pathlib import Path
path = Path.home()/'.config/silent-space/staging.json'
config = json.loads(path.read_text())
stamp = datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S%fZ')
shutil.copy2(path, path.with_name('staging-before-wallets-'+stamp+'.json'))
config['allow_manual_transfers'] = True
with tempfile.NamedTemporaryFile('w', dir=path.parent, delete=False) as f:
    json.dump(config, f, indent=2)
    temporary = Path(f.name)
os.chmod(temporary, 0o600)
os.replace(temporary, path)
print('Manual-wallet opt-in enabled; card remains disabled.')
PY
